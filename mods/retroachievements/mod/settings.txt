# The RetroAchievements tab of Settings (mod interface 7).
tab "RetroAchievements"
heading "Sign in"
option ra.user text "Username" 20 "" word ?ra.who=
option ra.pass secret "Password" 64 ?ra.who=
help ra.pass "Used only to sign in. Not stored."
button ra.signin "Sign in" ?ra.who=
show ra.who "Signed in as"
button ra.signout "Sign out" ?ra.who!=
option ra.hardcore bool "Hardcore mode" on
help ra.hardcore "Takes effect on next launch. Unavailable while gameplay-altering mods or debug switches are active."
show ra.why "Hardcore is off"
show ra.queue "Unlocks"
